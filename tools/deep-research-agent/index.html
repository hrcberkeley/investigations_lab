How it works

Perry runs entirely in your browser using WebGPU. It generates a research plan, then runs four specialized search passes — triaging results by relevance before fetching — extracts structured notes, and synthesizes them into a cited report. Nothing leaves your machine.

Default research angles

Perry searches through four general-purpose lenses: Overview & Background, Data & Statistics, Expert Analysis, and News & Reports. Each angle generates targeted queries and extracts different kinds of evidence.

You can customize angles, report sections, and extraction prompts in Advanced Settings. Use Collection mode to build a structured table instead of a report.

Setting up your Cloudflare Worker

Perry routes Brave Search and webpage fetching through a single worker you host for free on Cloudflare. Takes about 2 minutes.

Step 1 — Go to workers.cloudflare.com, log in, and click Create a Worker.

Step 2 — Replace all the default code with the script below. Before deploying, edit the ALLOWED_ORIGINS array to include your app's domain (the line is marked with a comment).

const BRAVE_SEARCH_BASE = "https://api.search.brave.com"; // Add every domain you serve this app from. const ALLOWED_ORIGINS = [ "http://localhost:8000", "http://127.0.0.1:8000", "http://localhost:5173", "http://127.0.0.1:5173", "http://localhost:3000", "http://127.0.0.1:3000", // "https://yourdomain.com", ← uncomment and replace with your domain ]; const MAX_PROXY_BYTES = 1_000_000; const MAX_REDIRECTS = 3; const ALLOWED_PROXY_PROTOCOLS = new Set(["http:", "https:"]); const ALLOWED_PROXY_CONTENT_TYPES = [ "text/html", "text/plain", "application/xhtml+xml", "application/xml", "text/xml", ]; export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { if (!isAllowedOrigin(request)) return new Response("Forbidden", { status: 403 }); return corsResponse(null, 204, request); } if (!isAllowedOrigin(request)) return new Response("Forbidden", { status: 403 }); if (request.method !== "GET") return corsResponse("Method Not Allowed", 405, request); const url = new URL(request.url); // Webpage proxy: /proxy?url=https://example.com if (url.pathname.startsWith("/proxy")) { const targetUrl = url.searchParams.get("url"); if (!targetUrl) return corsResponse("Missing url parameter", 400, request); let parsedTarget; try { parsedTarget = new URL(targetUrl); } catch { return corsResponse("Invalid url parameter", 400, request); } if (!ALLOWED_PROXY_PROTOCOLS.has(parsedTarget.protocol)) return corsResponse("Unsupported URL protocol", 400, request); try { const response = await fetchProxyTarget(parsedTarget.toString()); const contentType = response.headers.get("Content-Type") || "text/html"; const normalized = contentType.split(";")[0].trim().toLowerCase(); if (!ALLOWED_PROXY_CONTENT_TYPES.includes(normalized)) return corsResponse("Unsupported content type", 415, request); const body = await readLimitedText(response, MAX_PROXY_BYTES); return corsResponse(body, response.status, request, { "Content-Type": contentType }); } catch (err) { return corsResponse(`Failed to fetch: ${err.message}`, 502, request); } } // Brave Search relay: /res/v1/web/search?q=... if (!url.pathname.startsWith("/res/v1/")) return corsResponse("Not Found", 404, request); const apiKey = request.headers.get("X-Subscription-Token"); if (!apiKey) return corsResponse(JSON.stringify({ error: "Missing X-Subscription-Token" }), 401, request, { "Content-Type": "application/json" }); const braveUrl = `${BRAVE_SEARCH_BASE}${url.pathname}${url.search}`; let braveResponse; try { braveResponse = await fetch(braveUrl, { method: "GET", headers: { Accept: "application/json", "Accept-Encoding": "gzip", "X-Subscription-Token": apiKey }, }); } catch (err) { return corsResponse(JSON.stringify({ error: err.message }), 502, request, { "Content-Type": "application/json" }); } const body = await braveResponse.text(); return corsResponse(body, braveResponse.status, request, { "Content-Type": braveResponse.headers.get("Content-Type") || "application/json", }); }, }; function corsResponse(body, status, request, extraHeaders = {}) { const origin = request.headers.get("Origin"); return new Response(body, { status, headers: { "Access-Control-Allow-Origin": origin, "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, X-Subscription-Token", "Access-Control-Max-Age": "86400", "Vary": "Origin", ...extraHeaders, }, }); } function isAllowedOrigin(request) { const origin = request.headers.get("Origin"); return origin && ALLOWED_ORIGINS.includes(origin); } async function fetchProxyTarget(url, redirectsRemaining = MAX_REDIRECTS) { const response = await fetch(url, { method: "GET", redirect: "manual", headers: { "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,text/plain;q=0.8", "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36", }, }); if ([301, 302, 303, 307, 308].includes(response.status)) { if (redirectsRemaining <= 0) throw new Error("Too many redirects"); const location = response.headers.get("Location"); if (!location) throw new Error("Redirect without location"); const nextUrl = new URL(location, url); if (!ALLOWED_PROXY_PROTOCOLS.has(nextUrl.protocol)) throw new Error("Redirected to unsupported protocol"); return fetchProxyTarget(nextUrl.toString(), redirectsRemaining - 1); } return response; } async function readLimitedText(response, maxBytes) { const reader = response.body?.getReader(); if (!reader) return ""; const chunks = []; let total = 0; while (true) { const { done, value } = await reader.read(); if (done) break; total += value.byteLength; if (total > maxBytes) throw new Error(`Response exceeds ${maxBytes} bytes`); chunks.push(value); } const body = new Uint8Array(total); let offset = 0; for (const chunk of chunks) { body.set(chunk, offset); offset += chunk.byteLength; } return new TextDecoder().decode(body); }
